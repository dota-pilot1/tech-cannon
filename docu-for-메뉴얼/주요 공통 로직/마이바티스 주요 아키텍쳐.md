# MyBatis 주요 아키텍처

## 설정

```yaml
# application.yml
mybatis:
  mapper-locations: classpath:mybatis/mapper/**/*.xml
  type-aliases-package: com.mapo.palantier
  configuration:
    map-underscore-to-camel-case: true   # DB: created_at → Java: createdAt 자동 변환
    default-fetch-size: 100
    default-statement-timeout: 30
    log-impl: org.apache.ibatis.logging.stdout.StdOutImpl
```

## Mapper Interface 패턴

```java
@Mapper
public interface TaskFolderMapper {
    List<TaskFolder> findAll();
    Optional<TaskFolder> findById(@Param("id") Long id);
    void insert(TaskFolder folder);
    void update(TaskFolder folder);
    void softDelete(@Param("id") Long id);  // deleted_at 세팅
}
```

규칙:
- `@Mapper` 어노테이션 필수
- 복수 파라미터는 `@Param("name")` 사용
- 단일 엔티티 조회는 `Optional<T>` 반환
- 목록 조회는 `List<T>` 반환

## XML Mapper 패턴

```xml
<mapper namespace="com.mapo.palantier.task.folder.TaskFolderMapper">

    <!-- ResultMap: DB 컬럼 → Java 필드 매핑 -->
    <resultMap id="TaskFolderResultMap" type="com.mapo.palantier.task.folder.TaskFolder">
        <id property="id" column="id"/>
        <result property="parentId" column="parent_id"/>
        <result property="name" column="name"/>
        <result property="sortOrder" column="sort_order"/>
        <result property="createdBy" column="created_by"/>
        <result property="createdAt" column="created_at"/>
        <result property="updatedAt" column="updated_at"/>
    </resultMap>

    <!-- 목록 조회 -->
    <select id="findAll" resultMap="TaskFolderResultMap">
        SELECT id, parent_id, name, sort_order, created_by, created_at, updated_at
        FROM task_folders
        WHERE deleted_at IS NULL
        ORDER BY parent_id NULLS FIRST, sort_order
    </select>

    <!-- 단건 조회 -->
    <select id="findById" resultMap="TaskFolderResultMap">
        SELECT id, parent_id, name, sort_order, created_by, created_at, updated_at
        FROM task_folders
        WHERE id = #{id} AND deleted_at IS NULL
    </select>

    <!-- 삽입 (auto increment PK 반환) -->
    <insert id="insert" useGeneratedKeys="true" keyProperty="id">
        INSERT INTO task_folders (parent_id, name, sort_order, created_by)
        VALUES (#{parentId}, #{name}, #{sortOrder}, #{createdBy})
    </insert>

    <!-- 수정 -->
    <update id="update">
        UPDATE task_folders
        SET name = #{name}, parent_id = #{parentId},
            sort_order = #{sortOrder}, updated_at = CURRENT_TIMESTAMP
        WHERE id = #{id}
    </update>

    <!-- Soft Delete -->
    <update id="softDelete">
        UPDATE task_folders
        SET deleted_at = CURRENT_TIMESTAMP, updated_at = CURRENT_TIMESTAMP
        WHERE id = #{id}
    </update>
</mapper>
```

## Collection (1:N) 패턴

게시글에 블록 목록을 함께 조회할 때:

```xml
<resultMap id="TaskPostWithBlocksResultMap" type="com.mapo.palantier.task.post.TaskPost">
    <id property="id" column="id"/>
    <result property="title" column="title"/>
    <collection property="blocks" ofType="com.mapo.palantier.task.block.TaskBlock">
        <id property="id" column="block_id"/>
        <result property="blockType" column="block_type"
                typeHandler="org.apache.ibatis.type.EnumTypeHandler"
                javaType="com.mapo.palantier.task.block.BlockType"/>
        <result property="content" column="content"/>
        <result property="sortOrder" column="sort_order"/>
    </collection>
</resultMap>

<select id="findByIdWithBlocks" resultMap="TaskPostWithBlocksResultMap">
    SELECT p.id, p.title, p.author_id,
           b.id as block_id, b.block_type, b.content, b.sort_order
    FROM task_posts p
    LEFT JOIN task_blocks b ON p.id = b.post_id
    WHERE p.id = #{id} AND p.deleted_at IS NULL
    ORDER BY b.sort_order
</select>
```

## Enum TypeHandler

커스텀 Enum 타입은 `javaType`을 명시해야 함:

```xml
<!-- XML에서 직접 지정 -->

        typeHandler="org.apache.ibatis.type.EnumTypeHandler"
```

## Soft Delete 패턴

물리 삭제 대신 `deleted_at` 타임스탬프 기록:

```sql
-- Soft Delete
UPDATE task_folders
SET deleted_at = CURRENT_TIMESTAMP, updated_at = CURRENT_TIMESTAMP
WHERE id = #{id}

-- 조회 시 항상 deleted_at IS NULL 조건 포함
SELECT * FROM task_folders WHERE deleted_at IS NULL
```

## 동적 SQL

```xml
<!-- foreach: IN 절 동적 생성 -->
<select id="findByIds" resultMap="ItemResultMap">
    SELECT * FROM subutai_github_items
    WHERE id IN
    <foreach item="id" collection="ids" open="(" separator="," close=")">
        #{id}
    </foreach>
</select>

<!-- if: 조건부 WHERE -->
<select id="findByFolderId">
    SELECT * FROM task_posts
    WHERE deleted_at IS NULL
    <if test="folderId != null">
        AND folder_id = #{folderId}
    </if>
    ORDER BY updated_at DESC
</select>
```

## 재귀 CTE (PostgreSQL)

메뉴 트리, 폴더 계층 등에 사용:

```xml
<select id="findMenuTreeByRole" resultMap="MenuResultMap">
    WITH RECURSIVE menu_tree AS (
        SELECT * FROM menus WHERE parent_id IS NULL AND is_active = true
        UNION ALL
        SELECT m.* FROM menus m
        INNER JOIN menu_tree mt ON m.parent_id = mt.id
        WHERE m.is_active = true
    )
    SELECT * FROM menu_tree
    WHERE required_role IS NULL OR required_role = #{role}
    ORDER BY parent_id NULLS FIRST, order_num
</select>
```
